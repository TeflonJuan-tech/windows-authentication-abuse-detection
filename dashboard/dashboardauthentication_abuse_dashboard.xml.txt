      <table>
        <title>Potential Password Spray Activity</title>
        <search>
          <query>index=main sourcetype=XmlWinEventLog:Security "&lt;EventID&gt;4625&lt;/EventID&gt;"
| rex field=_raw "Data Name='TargetUserName'&gt;(?&lt;target_user&gt;[^&lt;]+)&lt;"
| rex field=_raw "Data Name='IpAddress'&gt;(?&lt;src_ip&gt;[^&lt;]+)&lt;"
| eval src_ip=coalesce(src_ip,"unknown")
| bin _time span=10m
| stats count as failures dc(target_user) as distinct_users values(target_user) as users by _time src_ip
| where distinct_users &gt;= 3 AND failures &gt;= 3
| sort - failures</query>
